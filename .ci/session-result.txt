  25 lemma volume_rotationalProfile_Ioo (d : ℕ) {a b : ℝ} (hab : a ≤ b)
  26     {g : ℝ → ℝ} (hg : Continuous g)
  27     (hg0 : ∀ t ∈ Set.Icc a b, 0 ≤ g t) :
  28     (volume : Measure (EuclideanSpace ℝ (Fin (d + 2))))
  29         (rotationalProfile (d + 1) (Set.Ioo a b) g) =
  30       ENNReal.ofReal (∫ t in a..b, g t ^ (d + 1)) *
  31         unitBallVolume (d + 1) := by
  32   rw [volume_rotationalProfile (d + 1) measurableSet_Ioo hg.measurable]
  33   simp_rw [volume_ball_eq_pow_mul_unit d]
  34   have hm : Measurable (fun t : ℝ => ENNReal.ofReal (g t) ^ (d + 1)) := by
  35     fun_prop
  36   rw [MeasureTheory.lintegral_mul_const (μ := volume.restrict (Set.Ioo a b))
  37     (unitBallVolume (d + 1)) hm]
  38   congr 1
  39   have hpow :
  40       (∫⁻ t in Set.Ioo a b, ENNReal.ofReal (g t) ^ (d + 1)) =
  41         ∫⁻ t in Set.Ioo a b, ENNReal.ofReal (g t ^ (d + 1)) := by
  42     apply setLIntegral_congr_fun measurableSet_Ioo
  43     intro t ht
  44     exact (ENNReal.ofReal_pow (hg0 t ⟨ht.1.le, ht.2.le⟩) (d + 1)).symm
  45   rw [hpow]
  46   rw [Measure.restrict_congr_set Ioo_ae_eq_Ioc]
  47   rw [← ENNReal.ofReal_integral_eq_lintegral_ofReal
  48     (hg.pow (d + 1)).intervalIntegrable.1]
  49   · rw [intervalIntegral.integral_of_le hab]
  50   · filter_upwards [ae_restrict_mem measurableSet_Ioc] with t ht
  51     exact pow_nonneg (hg0 t ⟨ht.1.le, ht.2⟩) _
  52 
  53 end
