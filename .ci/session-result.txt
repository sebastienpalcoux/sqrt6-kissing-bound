  58     (∫ t in (0 : ℝ)..a, (t / s3) ^ (k + 1)) =
  59       s3 / ((k + 2) * 2 ^ (k + 2)) := by
  60   rw [integral_div_mul_pow (k + 1) s3_ne]
  61   exact lower_power_value k
  62 
  63 lemma upper_profile_integral (k : ℕ) :
  64     (∫ t in a..1, (upperSlope * (1 - t)) ^ (k + 1)) =
  65       (2 - s3) / ((k + 2) * 2 ^ (k + 2)) := by
  66   rw [integral_const_mul_sub_pow (k + 1)]
  67   exact upper_power_value k

  69 lemma bicone_profile_integral (k : ℕ) :
  70     (∫ t in (0 : ℝ)..a, (t / s3) ^ (k + 1)) +
  71         (∫ t in a..1, (upperSlope * (1 - t)) ^ (k + 1)) =
  72       1 / ((k + 2) * 2 ^ (k + 1)) := by
  73   rw [lower_profile_integral, upper_profile_integral]

  84 lemma volumeReal_lowerBicone (k : ℕ) :
  85     (volume : Measure (EuclideanSpace ℝ (Fin (k + 2)))).real (lowerBicone k) =
  86       (s3 / ((k + 2) * 2 ^ (k + 2))) * unitBallVolumeReal (k + 1) := by
  87   have h := volumeReal_rotationalProfile_Ioo k
  88     (a := (0 : ℝ)) (b := a) (g := fun t => t / s3)
  89     (by positivity) (continuous_id.div_const _)

  91       intro t ht
  92       exact div_nonneg ht.1 s3_pos.le)
  93   simpa [lowerBicone, lower_profile_integral] using h

  98         unitBallVolumeReal (k + 1) := by
  99   have h := volumeReal_rotationalProfile_Ioo k
 100     (a := a) (b := (1 : ℝ)) (g := fun t => upperSlope * (1 - t))

 104       intro t ht
 105       have : 0 ≤ 1 - t := sub_nonneg.mpr ht.2
