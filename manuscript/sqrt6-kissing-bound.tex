\author{S\'ebastien Palcoux}

\begin{abstract}
Let $\tau_n$ denote the kissing number in Euclidean dimension $n$. We prove that
\[
\tau_n\leq (\sqrt{6})^n
\qquad(n\geq 1).
\]
The proof packs explicit bodies of revolution inside the unit ball. An inscribed bicone gives the estimate in dimension two and every dimension at least five; slightly larger bodies suffice in dimensions three and four. The body volumes are obtained by one-dimensional integration. Since $\tau_2=6$, the base $\sqrt{6}$ is optimal for a bound of this form valid in every dimension. The argument, including its geometric, analytic, and numerical steps, is formalized in Lean~4/Mathlib.
\end{abstract}

\section{Statement of the bound}

A \emph{kissing configuration} in $\R^n$ is a finite set $X$ satisfying
\begin{equation}\label{eq:configuration}
 \|x\|=1\quad(x\in X),
 \qquad
 \langle x,z\rangle\leq\frac12\quad(x,z\in X,\ x\ne z).
\end{equation}
Multiplying its vectors by two gives the centers of nonoverlapping unit balls touching the unit ball at the origin. We write $\tau_n$ for the largest cardinality of a kissing configuration in $\R^n$. The finite bound proved below also establishes the existence of this maximum.

\begin{theorem}\label{thm:main}
For every integer $n\geq1$,
\[
\tau_n\leq (\sqrt{6})^n.
\]
\end{theorem}

\begin{corollary}\label{cor:base}
The least real number $\alpha>0$ such that
\[
\tau_n\leq\alpha^n
\qquad\text{for every }n\geq1
\]
is $\sqrt{6}$. Equivalently,
\[
\sup_{n\geq1}\tau_n^{1/n}=\sqrt{6}.
\]
\end{corollary}

The point is the optimal base in a bound that holds in every dimension with no additional prefactor. No claim of asymptotic optimality is intended.

\section{Proof}

Throughout, put $s=\sqrt3$ and $a=s/2$. Let $B_n$ be the open unit ball in $\R^n$, and write $V_n=\vol(B_n)$ for its Lebesgue volume.

\subsection{Disjoint cones and a volume comparison}

For a unit vector $x$, define the open cone
\[
 C_x=\{y\in\R^n:\langle x,y\rangle>a\|y\|\}.
\]
Thus $C_x$ has axis $x$ and half-angle $\pi/6$.

\begin{lemma}\label{lem:packing}
Let $X$ be a kissing configuration in $\R^n$, where $n\geq2$, and let $e$ be a fixed unit vector. If $K$ is a measurable subset of $C_e\cap B_n$ with volume $q$, then
\begin{equation}\label{eq:packing}
 |X|q\leq V_n.
\end{equation}
\end{lemma}

\begin{proof}
We first check that $C_x\cap C_z=\varnothing$ for distinct $x,z\in X$. Otherwise, normalizing a vector in the intersection gives a unit vector $u$ with
\[
 b=\langle x,u\rangle>a,
 \qquad c=\langle z,u\rangle>a.
\]
Write $x=bu+p$ and $z=cu+r$, where $p,r\perp u$. Then
\[
 \|p\|^2=1-b^2<\frac14,
 \qquad \|r\|^2=1-c^2<\frac14.
\]
Cauchy--Schwarz now gives
\[
 \langle x,z\rangle=bc+\langle p,r\rangle
 \geq bc-\|p\|\|r\|>\frac34-\frac14=\frac12,
\]
contrary to \eqref{eq:configuration}.

For each $x\in X$, choose an orthogonal transformation sending $e$ to $x$. Its image $K_x$ of $K$ lies in $C_x\cap B_n$ and has volume $q$. These measurable images are pairwise disjoint, so finite additivity and inclusion in $B_n$ give \eqref{eq:packing}.
\end{proof}

We construct the bodies in orthogonal coordinates $(t,v)\in\R\times\R^{n-1}$, with $e=(1,0)$. In these coordinates,
\begin{equation}\label{eq:coordinateconditions}
 (t,v)\in C_e
 \quad\Longleftrightarrow\quad t>0\ \text{and}\ 3\|v\|^2<t^2,
 \qquad
 (t,v)\in B_n
 \quad\Longleftrightarrow\quad t^2+\|v\|^2<1.
\end{equation}
For a continuous nonnegative function $g$ on $[b,c]$, Fubini's theorem and scaling of Lebesgue measure give
\begin{equation}\label{eq:profilevolume}
 \vol\{(t,v):b<t<c,\ \|v\|<g(t)\}
 =V_{n-1}\int_b^c g(t)^{n-1}\,dt.
\end{equation}
Indeed, the section at $t$ is an $(n-1)$-dimensional ball of radius $g(t)$. Every body used below is a finite union of such measurable sets, with disjoint intervals for the first coordinate.

\subsection{The bicone bound}

For $n\geq2$, consider
\begin{align*}
 L_n&=\{(t,v):0<t<a,\ \|v\|<t/s\},\\
 U_n&=\{(t,v):a<t<1,\ \|v\|<(2+s)(1-t)\}.
\end{align*}
Their union $K_n=L_n\cup U_n$ is a bicone, with the common section at $t=a$ omitted. Both profiles have radius $1/2$ at that section.

\begin{lemma}\label{lem:bicone}
The body $K_n$ lies in $C_e\cap B_n$ and has volume
\begin{equation}\label{eq:biconevolume}
 \vol(K_n)=\frac{V_{n-1}}{n2^{n-1}}.
\end{equation}
Consequently every kissing configuration $X\subset\R^n$ satisfies
\begin{equation}\label{eq:biconebound}
 |X|\leq n2^{n-1}\frac{V_n}{V_{n-1}}.
\end{equation}
\end{lemma}

\begin{proof}
For $(t,v)\in L_n$, one has $3\|v\|^2<t^2$ and
\[
 t^2+\|v\|^2<\frac43t^2<\frac43a^2=1.
\]
For the upper profile $h(t)=(2+s)(1-t)$, the identities
\[
 sh(a)=a,
 \qquad (2+s)^2(1-a)=1+a
\]
show, for $a<t<1$, that
\[
 sh(t)<t,
 \qquad h(t)^2<1-t^2.
\]
Here the first inequality compares a decreasing affine function with $t$, and the second follows by multiplying
$(2+s)^2(1-t)<1+t$ by $1-t>0$.
Thus \eqref{eq:coordinateconditions} puts both pieces inside $C_e\cap B_n$.

The two integrals in \eqref{eq:profilevolume} are
\begin{align*}
 \int_0^a(t/s)^{n-1}\,dt&=\frac{s}{n2^n},\\
 \int_a^1\bigl((2+s)(1-t)\bigr)^{n-1}\,dt
 &=\frac{2-s}{n2^n}.
\end{align*}
For the second equality, use $(2+s)(1-a)=1/2$. Adding the integrals proves \eqref{eq:biconevolume}; Lemma~\ref{lem:packing} then proves \eqref{eq:biconebound}.
\end{proof}

We use the standard unit-ball volume formula and its recurrence:
\begin{equation}\label{eq:ballrecurrence}
 V_d=\frac{\pi^{d/2}}{\Gamma(d/2+1)},
 \qquad
 V_{d+2}=\frac{2\pi}{d+2}V_d
 \quad(d\geq1).
\end{equation}
The recurrence follows directly from $\Gamma(z+1)=z\Gamma(z)$. In particular,
\begin{equation}\label{eq:smallvolumes}
\begin{gathered}
 V_1=2,\qquad V_2=\pi,\qquad V_3=\frac{4\pi}{3},
 \qquad V_4=\frac{\pi^2}{2},\\
 V_5=\frac{8\pi^2}{15},\qquad
 V_6=\frac{\pi^3}{6},\qquad
 V_7=\frac{16\pi^3}{105}.
\end{gathered}
\end{equation}
The estimates for $\pi$ needed here are implied by $0<\pi<22/7$.

\begin{lemma}\label{lem:high}
For every integer $n\geq6$,
\[
 V_n\leq V_{n-1},
 \qquad n2^{n-1}\leq(\sqrt6)^n.
\]
\end{lemma}

\begin{proof}
The first inequality holds at $n=6$, since
$V_6/V_5=5\pi/16<1$, and at $n=7$, since
$V_7/V_6=32/35<1$. If it holds two dimensions earlier, then \eqref{eq:ballrecurrence} gives
\[
 V_n=\frac{2\pi}{n}V_{n-2}
 \leq\frac{2\pi}{n}V_{n-3}
 \leq\frac{2\pi}{n-1}V_{n-3}=V_{n-1}.
\]
This proves the first assertion by induction separately on even and odd dimensions.

For the second assertion, the initial cases are
\[
 6\cdot2^5=192<216=(\sqrt6)^6,
 \qquad
 7\cdot2^6=448<216\sqrt6=(\sqrt6)^7;
\]
the latter inequality follows by squaring $56/27<\sqrt6$. If the assertion holds at $n-2$ and $n\geq8$, then $4n\leq6(n-2)$ gives
\[
 n2^{n-1}=4n2^{n-3}
 \leq6(n-2)2^{n-3}
 \leq6(\sqrt6)^{n-2}=(\sqrt6)^n.
\]
\end{proof}

Together, Lemmas~\ref{lem:bicone} and~\ref{lem:high} prove the required cardinality bound for every $n\geq6$. The bicone also suffices in dimensions two and five. Indeed, \eqref{eq:biconebound} and \eqref{eq:smallvolumes} give
\begin{equation}\label{eq:twofive}
 n=2:\quad |X|\leq2\pi<7,
 \qquad
 n=5:\quad |X|\leq\frac{256}{3}<86.
\end{equation}
Since $|X|$ is an integer, these bounds imply respectively
$|X|\leq6=(\sqrt6)^2$ and
$|X|\leq85<36\sqrt6=(\sqrt6)^5$.
The last strict inequality follows from $85^2<36^2\cdot6$.

\subsection{Dimension three}

In dimension three, retain $L_3$ and replace the upper part of the bicone by
\[
 W_3=\{(t,v):a<t<1,\ \|v\|<\sqrt{1-t^2}\}.
\]
This body is in $B_3$ by construction. Moreover, $t>a$ implies
$3(1-t^2)<t^2$, so \eqref{eq:coordinateconditions} places $W_3$ in $C_e$ as well.
By \eqref{eq:profilevolume},
\begin{align*}
 \vol(L_3\cup W_3)
 &=V_2\left(\frac{s}{24}+\int_a^1(1-t^2)\,dt\right)\\
 &=\pi\left(\frac{s}{24}+\frac23-\frac{3s}{8}\right)
 =\frac{(2-s)\pi}{3}.
\end{align*}
Since $s<26/15$, as follows by squaring, this volume is strictly larger than $4\pi/45=V_3/15$. Lemma~\ref{lem:packing} therefore yields $|X|<15$, hence
\begin{equation}\label{eq:three}
 |X|\leq14<6\sqrt6=(\sqrt6)^3.
\end{equation}
Here the final comparison follows from $14^2<36\cdot6$.

\subsection{Dimension four}

For dimension four, a polynomial upper profile makes the volume calculation elementary. Set
\[
 p(t)=(1-t)(14t+2-6s),
 \qquad
 W_4=\{(t,v):a<t<1,\ \|v\|<p(t)\}.
\]
We shall prove that $L_4\cup W_4$ lies in $C_e\cap B_4$ and has volume greater than $V_4/37$.

First, $p(a)=1/2$, $p(1)=0$, and $p(t)\geq0$ on $[a,1]$, since
$14t+2-6s\geq s+2>0$. To check containment in the ball, define
\[
 Q(t)=196t^2-(140+70s)t-48+74s.
\]
A direct factorization using $s^2=3$ gives
\begin{equation}\label{eq:polynomialcontainment}
 1-t^2-p(t)^2=(1-t)(t-a)Q(t).
\end{equation}
For $t\geq a$, the identity
\[
 Q(t)=4s-6+(t-a)\bigl(196(t+a)-140-70s\bigr)
\]
shows that $Q(t)>0$: indeed $s>3/2$, and the expression in parentheses is at least $126s-140>0$. Thus
\[
 p(t)^2\leq1-t^2\qquad(a\leq t\leq1).
\]
Every point of $W_4$ therefore lies in $B_4$. As in dimension three, $t>a$ then implies $3\|v\|^2<t^2$, so $W_4\subset C_e$.

It remains to compute its volume. Expanding $p(t)^3$ and integrating its monomials gives
\begin{equation}\label{eq:fourintegral}
 \int_a^1p(t)^3\,dt=\frac{33169-19150s}{40}.
\end{equation}
For an explicit evaluation, put $\delta=1-a$ and $A=16-6s$. The substitution $u=1-t$ gives $p(t)=u(A-14u)$, and the left side of \eqref{eq:fourintegral} equals
\[
 \frac{A^3\delta^4}{4}-\frac{42A^2\delta^5}{5}
       +98A\delta^6-392\delta^7.
\]
Substituting $\delta=1-s/2$ and reducing powers with $s^2=3$ yields \eqref{eq:fourintegral}. Adding the lower part, whose profile integral is $s/64$, gives
\begin{equation}\label{eq:fourvolume}
 \vol(L_4\cup W_4)=T V_3,
 \qquad
 T=\frac{265352-153195s}{320}.
\end{equation}

All that is needed is a rational lower estimate for $T$. The inequality
$s<1351/780$ follows from $1351^2-3\cdot780^2=1$. Consequently
\[
 T>\frac{541}{16640}>\frac{33}{1036}
   =\frac{3}{296}\frac{22}{7}>\frac{3\pi}{296}.
\]
Using \eqref{eq:smallvolumes}, we obtain
\[
 \vol(L_4\cup W_4)>
 \frac{3\pi}{296}\frac{4\pi}{3}
 =\frac{\pi^2}{74}=\frac{V_4}{37}.
\]
Lemma~\ref{lem:packing} now yields $|X|<37$. Integrality gives
\begin{equation}\label{eq:four}
 |X|\leq36=(\sqrt6)^4.
\end{equation}

\subsection{Completion and optimality}

\begin{proof}[Proof of Theorem~\ref{thm:main}]
Let $X$ be any kissing configuration in $\R^n$. For $n=1$, every unit vector is $1$ or $-1$, so $|X|\leq2<\sqrt6$. Dimensions two and five are covered by \eqref{eq:twofive}, dimensions three and four by \eqref{eq:three} and \eqref{eq:four}, and every $n\geq6$ by Lemmas~\ref{lem:bicone} and~\ref{lem:high}. Thus $|X|\leq(\sqrt6)^n$ in every positive dimension.

The realizable cardinalities form a nonempty subset of
\[
 \{0,\ldots,\lfloor(\sqrt6)^n\rfloor\}.
\]
This finite set of cardinalities has a largest element, realized by a configuration. This element is $\tau_n$, and it satisfies the same bound.
\end{proof}

\begin{proof}[Proof of Corollary~\ref{cor:base}]
The six points
\[
 (1,0),\quad (\tfrac12,\tfrac{s}{2}),\quad
 (-\tfrac12,\tfrac{s}{2}),\quad (-1,0),\quad
 (-\tfrac12,-\tfrac{s}{2}),\quad (\tfrac12,-\tfrac{s}{2})
\]
are distinct unit vectors, and the inner product of any two distinct members is at most $1/2$. They form a kissing configuration. Theorem~\ref{thm:main} gives the reverse bound, so $\tau_2=6$.

The base $\sqrt6$ works by Theorem~\ref{thm:main}. Any positive universal base $\alpha$ must satisfy $6=\tau_2\leq\alpha^2$, hence $\alpha\geq\sqrt6$. Likewise each $\tau_n^{1/n}\leq\sqrt6$, with equality at $n=2$, which proves the supremum statement.
\end{proof}
