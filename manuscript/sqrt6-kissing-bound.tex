\title{A Universal Square-Root-of-Six Bound for Kissing Numbers}
\author{S\'ebastien Palcoux}
\date{November 18, 2026}

\begin{document}
\maketitle

\begin{abstract}
The least positive number $\alpha$ for which the Euclidean kissing numbers satisfy
$\tau_n\leq\alpha^n$ in every positive dimension is $\sqrt6$. We show how this follows from a classical spherical-cap estimate and known low-dimensional bounds, then give two elementary proofs that require no tables of kissing numbers. A recurrence for cap areas gives strict inequality in every dimension except two. A second proof packs explicit bodies of revolution into the unit ball. The Lean~4/Mathlib certificate registered in Palomar uses this volume construction to prove the universal bound, the planar value $\tau_2=6$, and the optimality of the base.
\end{abstract}

\section{The universal base}

A \emph{kissing configuration} in $\R^n$ is a finite set $X$ satisfying
\begin{equation}\label{eq:configuration}
 \|x\|=1\quad(x\in X),\qquad
 \langle x,z\rangle\leq\frac12\quad(x,z\in X,\ x\ne z).
\end{equation}
Multiplying its vectors by two gives the centers of nonoverlapping unit balls touching the unit ball at the origin. We write $\tau_n$ for the largest cardinality of a kissing configuration in $\R^n$. The bounds below also establish the existence of this maximum.

\begin{theorem}\label{thm:main}
For every integer $n\geq1$,
\[
\tau_n\leq(\sqrt6)^n.
\]
\end{theorem}

\begin{corollary}\label{cor:base}
The least real number $\alpha>0$ such that $\tau_n\leq\alpha^n$ for every $n\geq1$ is $\sqrt6$. Equivalently,
\[
\sup_{n\geq1}\tau_n^{1/n}=\sqrt6.
\]
\end{corollary}

The planar regular hexagon forces every such $\alpha$ to be at least $\sqrt6$. The issue is whether that same base works with prefactor one in every dimension. The question was raised in 2017~\cite{MO-question}, motivated by a result of Bakshi, Das, Liu, and Ren~\cite[Theorem~4.1]{BDLR}. For a finite-index irreducible inclusion $N\subset M$ of type $\mathrm{II}_1$ factors, the number of minimal intermediate subfactors is at most $\tau_d$, where $d=\dim_{\mathbb C}(N'\cap M_1)$ and $M_1$ is the basic construction. Theorem~\ref{thm:main} thus gives the explicit bound $6^{d/2}$. Cohn~\cite{Cohn-answer} anticipated the value $\sqrt6$ and outlined an approach using the finite-dimensional bound of Kabatiansky and Levenshtein.

Throughout, put $s=\sqrt3$ and $a=s/2$. For a unit vector $x$, let
\[
 C_x=\{y\in\R^n:\langle x,y\rangle>a\|y\|\}.
\]
This is the open cone of half-angle $\pi/6$ about $x$. Its intersection with the unit sphere is the corresponding open spherical cap.

\begin{lemma}\label{lem:separation}
For distinct members $x,z$ of a kissing configuration, $C_x\cap C_z=\varnothing$.
\end{lemma}

\begin{proof}
A vector in the intersection could be normalized to a unit vector $u$ satisfying
\[
 s<\langle x+z,u\rangle\leq\|x+z\|
 =\sqrt{2+2\langle x,z\rangle}\leq\sqrt3=s,
\]
a contradiction.
\end{proof}

In dimension one the only unit vectors are $1$ and $-1$, so $\tau_1=2<\sqrt6$.

\paragraph{Deduction from published bounds.}
For $n\geq2$, write $\Omega_n(\theta)$ for the fraction of the area of $S^{n-1}$ covered by a cap of angular radius $\theta$. For $0<\theta<\pi/2$, the estimate of B\"or\"oczky and Wintsche~\cite[Corollary~3.2(i)]{BW}, also stated in~\cite[Lemma~3.4, (7)]{Naszodi}, gives
\[
 \Omega_n(\theta)\geq\frac{(\sin\theta)^{n-1}}{\sqrt{2\pi n}}.
\]
By Lemma~\ref{lem:separation}, the open caps of angular radius $\pi/6$ about the points of $X$ are disjoint. Hence $|X|\leq2^{n-1}\sqrt{2\pi n}$. For $n\geq6$ this is strictly smaller than $6^{n/2}$. Squaring reduces the comparison to $n\pi/2<(3/2)^n$. It holds at $n=6$ since $3\pi<(3/2)^6$, and $(3/2)^n/n$ increases for $n>2$. In dimensions one and two the elementary bounds are $2$ and $6$ (in the plane the angles between consecutive directions are at least $\pi/3$). For dimensions three through five, the upper bounds $12$, $24$, and $45$ in Bachoc and Vallentin~\cite[Table~5.1]{BV} suffice. Thus results already available before the 2017 question imply Theorem~\ref{thm:main}. The proofs below derive all the needed low-dimensional estimates directly.

\section{First proof: a spherical-cap recurrence}\label{sec:cap}

For $n\geq2$ and $0<\theta<\pi/2$, polar coordinates about the cap center give
\begin{equation}\label{eq:capfraction}
 \Omega_n(\theta)=\frac{I_{n-2}(\theta)}{J_{n-2}},\qquad
 I_m(\theta)=\int_0^\theta\sin^m t\,dt,\qquad J_m=\int_0^\pi\sin^m t\,dt.
\end{equation}
Indeed, the polar area element has a factor $\sin^{n-2}t$, and the transverse area cancels. Put $c_n=\Omega_n(\pi/6)$. Disjointness gives
\begin{equation}\label{eq:cappacking}
 |X|c_n\leq1.
\end{equation}

\begin{lemma}\label{lem:caprecurrence}
For $n\geq2$ and $0<\theta<\pi/2$,
\[
 \frac{\Omega_{n+2}(\theta)}{\Omega_n(\theta)}
   >\frac{n}{n+1}\sin^2\theta.
\]
In particular, $c_{n+2}>\dfrac{n}{4(n+1)}c_n\geq\dfrac16c_n$.
\end{lemma}

\begin{proof}
Put $m=n-2$ and $h=\sin\theta$. Substituting $u=\sin t$ gives
$I_m(\theta)=\int_0^h u^m w(u)\,du$, with $w(u)=(1-u^2)^{-1/2}$.
Give $[0,h]$ the probability measure $d\mu(u)=(m+1)u^m\,du/h^{m+1}$ and denote its expectation by $E$. Both $u^2$ and $w(u)$ are strictly increasing. Thus
\[
 2\bigl(E[u^2w]-E[u^2]E[w]\bigr)
  =\iint (u^2-v^2)\bigl(w(u)-w(v)\bigr)\,d\mu(u)\,d\mu(v)>0,
\]
since the integrand is positive off the diagonal. It follows that
\[
 \frac{I_{m+2}(\theta)}{I_m(\theta)}
 =\frac{E[u^2w]}{E[w]}>E[u^2]=h^2\frac{m+1}{m+3}.
\]
Integration by parts gives $J_{m+2}/J_m=(m+1)/(m+2)$. Dividing the preceding inequality by this ratio gives the result.
\end{proof}

\begin{proof}[First proof of Theorem~\ref{thm:main}]
Direct integration gives $c_2=1/6$ and $c_3=(2-s)/4>1/15$, since $s<26/15$ by squaring. From $c_2$ the strict recurrence gives $c_n>6^{-n/2}$ in every even dimension $n\geq4$. In dimension three, \eqref{eq:cappacking} gives $|X|<15$. Integrality yields $|X|\leq14<6\sqrt6$. To start the odd induction we use the stronger coefficient at $n=3$:
\[
 c_5>\tfrac3{16}c_3>\tfrac1{80}>6^{-5/2},
\]
where the last inequality follows from $80^2<6^5$. Thus $c_n>6^{-n/2}$ for every odd $n\geq5$ as well. Equation~\eqref{eq:cappacking} proves the result, with strict inequality for $n\ne2$. In each dimension the set of realizable cardinalities is nonempty and bounded in the nonnegative integers, so it has a greatest element.
\end{proof}

\section{Second proof: explicit Euclidean volumes}\label{sec:volume}

The second proof uses the bodies in the registered Lean certificate. For any unit direction $e$, radial integration identifies the fraction of the unit ball in $C_e$ with the cap fraction $c_n$ of Section~\ref{sec:cap}. The common radial factor $\int_0^1r^{n-1}\,dr$ cancels. To obtain a packing bound it suffices to inscribe a body of sufficiently large volume in that sector. We now do this using only Euclidean integration and Lemma~\ref{lem:separation}.

Let $B_n$ be the open unit ball and $V_n=\vol(B_n)$. Fix $e=(1,0)$ in orthogonal coordinates $(t,v)\in\R\times\R^{n-1}$. Rotating any measurable body $K\subset C_e\cap B_n$ toward the points of $X$ gives disjoint bodies of equal volume by Lemma~\ref{lem:separation}. Thus
\begin{equation}\label{eq:volpacking}
 |X|\vol(K)\leq V_n.
\end{equation}
For a continuous nonnegative profile $g$ on $[b,c]$, Fubini's theorem gives
\begin{equation}\label{eq:profilevolume}
 \vol\{(t,v):b<t<c,\ \|v\|<g(t)\}
 =V_{n-1}\int_b^c g(t)^{n-1}\,dt.
\end{equation}
The coordinate conditions for containment are
\begin{equation}\label{eq:coordinates}
 (t,v)\in C_e\iff t>0\text{ and }3\|v\|^2<t^2,
 \qquad (t,v)\in B_n\iff t^2+\|v\|^2<1.
\end{equation}

\subsection{A bicone and the dimensions \texorpdfstring{$n\geq5$}{n >= 5}}

Consider the union $K_n=L_n\cup U_n$, where
\begin{align*}
 L_n&=\{(t,v):0<t<a,\ \|v\|<t/s\},\\
 U_n&=\{(t,v):a<t<1,\ \|v\|<(2+s)(1-t)\}.
\end{align*}
It is a bicone of total height one and common base radius $1/2$, with the base section omitted. More precisely, each point is a strict convex combination of a point of the disk
\[
 D=\{(a,v):\|v\|<1/2\}
\]
with either $0$ or $e$. The disk lies in $C_e\cap B_n$, whereas $0,e$ satisfy the weak inequalities defining their closures. The triangle inequality therefore gives $K_n\subset C_e\cap B_n$.

By \eqref{eq:profilevolume}, the two cones have volumes
$aV_{n-1}/(n2^{n-1})$ and $(1-a)V_{n-1}/(n2^{n-1})$. Hence
\begin{equation}\label{eq:bicone}
 \vol(K_n)=\frac{V_{n-1}}{n2^{n-1}},\qquad
 |X|\leq A_n:=n2^{n-1}\frac{V_n}{V_{n-1}}.
\end{equation}
The standard ball-volume formula gives
\begin{equation}\label{eq:ballrecurrence}
 V_d=\frac{\pi^{d/2}}{\Gamma(d/2+1)},\qquad
 V_{d+2}=\frac{2\pi}{d+2}V_d,
\end{equation}
and consequently
\begin{equation}\label{eq:Anrecurrence}
 A_{n+2}=\frac{4(n+1)}{n}A_n\leq6A_n\qquad(n\geq2).
\end{equation}
The initial estimates
\[
 A_5=\frac{256}{3}<36\sqrt6=(\sqrt6)^5,\qquad
 A_6=60\pi<216=(\sqrt6)^6
\]
follow by squaring the first comparison and using $\pi<22/7$ for the second. Induction on each parity proves $|X|<(\sqrt6)^n$ for all $n\geq5$. In dimension two, the same bicone gives $|X|\leq A_2=2\pi<7$, so integrality gives $|X|\leq6$.

\subsection{Dimension three}

Retain $L_3$ and replace the upper cone by
\[
 W_3=\{(t,v):a<t<1,\ \|v\|<\sqrt{1-t^2}\}.
\]
Together with $L_3$ this fills the entire sector apart from the joining slice $t=a$, which has volume zero. Indeed, below $a$ the cone gives the smaller cross-section, and above $a$ the ball does. In particular $W_3$ lies in the cone because $t>a$ implies $3(1-t^2)<t^2$. Since $V_2=\pi$, formula~\eqref{eq:profilevolume} gives
\begin{align*}
 \vol(L_3\cup W_3)
 &=\pi\left(\frac{s}{24}+\int_a^1(1-t^2)\,dt\right)
 =\frac{(2-s)\pi}{3}
 >\frac{4\pi}{45}=\frac{V_3}{15}.
\end{align*}
Here $V_3=4\pi/3$, and the strict inequality uses $s<26/15$. Thus \eqref{eq:volpacking} yields $|X|<15$, hence $|X|\leq14<6\sqrt6$.

\subsection{Dimension four}\label{sec:four}

The bicone alone gives $A_4=12\pi>37$. To bring the packing bound below $37$ while keeping the volume integral polynomial, retain $L_4$ and raise the upper chord $q(t)=(2+s)(1-t)$ by a quadratic that vanishes at its endpoints. More precisely, set
\[
 p(t)=q(t)+14(1-t)(t-a)=(1-t)(14t+2-6s),\qquad
 W_4=\{(t,v):a<t<1,\ \|v\|<p(t)\}.
\]
The coefficient $14$ can be seen from the tangent at $a$. There $q$ meets the circular profile $\sqrt{1-t^2}$ at height $1/2$. For the raised chord with coefficient $k$ to stay below the circle on the right, its derivative must satisfy $-(2+s)+k(1-a)\leq-s$, or $k\leq 2/(1-a)=8+4s$. The largest integral candidate is $14$. The following calculation checks containment on the entire interval.

For $a\leq t\leq1$ we have $p(t)\geq0$, since $14t+2-6s\geq s+2>0$. Writing $u=t-a$, a factorization gives
\begin{equation}\label{eq:fourcontainment}
 1-t^2-p(t)^2
 =(1-t)(t-a)\bigl(4s-6+(126s-140)u+196u^2\bigr)\geq0.
\end{equation}
Indeed, $u\geq0$ and the constant and linear coefficients in the last factor are positive because $s>3/2$. Thus $W_4$ lies in the ball, and $t>a$ puts it in the cone as in dimension three.

A lower bound for the volume avoids cancellation in the exact integral. Write
$\delta=1-a$, $x=(t-a)/\delta$, and $b=14\delta^2$. Then
$p(t)=(1-x)(1/2+bx)$. The lower cone contributes $a/32$ to the profile integral in \eqref{eq:profilevolume}. For the upper piece use
$(1/2+bx)^3\geq 1/8+3bx/4$ on $[0,1]$. Integrating gives
\begin{align*}
 T:=\frac{\vol(L_4\cup W_4)}{V_3}
 &\geq\frac{a}{32}
    +\delta\int_0^1(1-x)^3\Bigl(\frac18+\frac{3b}{4}x\Bigr)\,dx\\
 &=\frac{a+\delta}{32}+\frac{3b\delta}{80}
  =\frac1{32}+\frac{21}{40}\delta^3 .
\end{align*}
Since $s<7/4$, we have $\delta>1/8$, so $T>1/32+\tfrac12(1/8)^3=33/1024$. Also $V_4/V_3=3\pi/8<33/28$ by $\pi<22/7$. The packing inequality~\eqref{eq:volpacking} therefore gives
\[
 |X|\leq\frac{V_4/V_3}{T}<\frac{33/28}{33/1024}=\frac{256}{7}<37.
\]
Thus $|X|\leq36=(\sqrt6)^4$.

\begin{proof}[Completion of the second proof of Theorem~\ref{thm:main}]
The volume estimates cover every $n\geq2$, and the dimension-one observation covers $n=1$. Thus every kissing configuration satisfies the claimed bound, independently of the first proof. As before, the finite nonempty set of realizable cardinalities has an attained maximum, which is $\tau_n$.
\end{proof}

\section{Sharpness and the equality case}

\begin{proof}[Proof of Corollary~\ref{cor:base}]
The vertices of a regular hexagon are
\[
 (1,0),\quad (\tfrac12,\tfrac{s}{2}),\quad
 (-\tfrac12,\tfrac{s}{2}),\quad (-1,0),\quad
 (-\tfrac12,-\tfrac{s}{2}),\quad (\tfrac12,-\tfrac{s}{2})
\]
with every inner product between distinct vertices at most $1/2$. These six unit vectors form a kissing configuration. Theorem~\ref{thm:main} gives the reverse bound, so $\tau_2=6$.

The base $\sqrt6$ works by the theorem. Conversely, any positive universal base $\alpha$ must satisfy $6=\tau_2\leq\alpha^2$, hence $\alpha\geq\sqrt6$. Likewise, $\tau_n^{1/n}\leq\sqrt6$ in every dimension, with equality at $n=2$, proving the supremum statement.
\end{proof}

\begin{corollary}\label{cor:unique}
Dimension two is the unique maximizer of $\tau_n^{1/n}$ among positive integers $n$.
\end{corollary}

\begin{proof}
The regular hexagon gives equality in dimension two, and the first proof of Theorem~\ref{thm:main} gives $\tau_n<6^{n/2}$ for every $n\ne2$.
\end{proof}

\section{Formal statements and provenance}\label{sec:lean}

They bound every finite kissing configuration, establish the existence of a greatest realizable cardinality, and give the planar value and equivalent formulations of the optimal base. The proof uses the Euclidean bodies of Section~\ref{sec:volume}, including the exceptional profiles in dimensions three and four. For the four-dimensional body the certificate evaluates the integral exactly, while the lower estimate in Section~\ref{sec:four} suffices here.
